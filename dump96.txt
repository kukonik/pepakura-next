 112: fn unfold_hinge(v1_3d: [f64; 3], v2_3d: [f64; 3], v3_3d: [f64; 3], v1_2d: [f64; 2], v2_2d: [f64; 2], sign: f64) -> [f64; 2] {
 113:     let u3 = sub3(v2_3d, v1_3d);
 114:     let w3 = sub3(v3_3d, v1_3d);
 115:     let u_len = len3(u3);
 116:     if u_len < 1e-12 { return v1_2d; }
 117:     let u_hat = [u3[0] / u_len, u3[1] / u_len, u3[2] / u_len];
 118:     let proj = dot3(w3, u_hat);
 119:     let perp = [w3[0] - u_hat[0] * proj, w3[1] - u_hat[1] * proj, w3[2] - u_hat[2] * proj];
 120:     let h = len3(perp);

 154: /// Диэдральный угол между гранями в градусах (0 = плоско, ср. Edge.angle в model.rs).
 155: fn edge_dihedral_deg(mesh: &Mesh, f1: usize, f2: usize) -> f64 {
 156:     dot3(face_normal(mesh, f1), face_normal(mesh, f2)).clamp(-1.0, 1.0).acos().to_degrees()
 157: }
 158:
 159: /// Почти-плоские рёбра (порог = верхняя ступень веса в dual_edge_weight).
 160: const FLAT_DEG: f64 = 10.0;
 161: // [SMART-CUT] Печатаемый размер детали: остров дерева больше MAX_PIECE_FACES
 162: // делится разрезом самого удобного ребра дерева (минимальный вес = самое
 163: // острое место), пока каждая часть не влезет в лимит. Осколки меньше
 164: // MIN_PIECE_FACES не образуются: кандидат обязан делить остров надвое.
 165: const MAX_PIECE_FACES: usize = 250;
 166: const MIN_PIECE_FACES: usize = 40;
 167:
 168: fn split_oversized_islands(mesh: &Mesh, dual: &mut [ForestEdge]) {
 169:     let nf = mesh.faces.len();
 170:     if nf == 0 { return; }
 171:     for _round in 0..nf {
 172:         let mut uf = Uf::new(nf);
 173:         for e in dual.iter() { if e.in_tree { uf.union(e.f1, e.f2); } }
 174:         let mut comp_size: HashMap<usize, usize> = HashMap::new();
 175:         for fi in 0..nf { *comp_size.entry(uf.find(fi)).or_insert(0) += 1; }
 176:         let mut big_root = usize::MAX;
 177:         let mut big_len = MAX_PIECE_FACES;
 178:         for (r, s) in comp_size.iter() {
 179:             if *s > big_len { big_len = *s; big_root = *r; }
 180:         }
 181:         if big_root == usize::MAX { return; }
 182:         let mut tedges: Vec<(usize, usize, usize)> = Vec::new();
 183:         for (idx, e) in dual.iter().enumerate() {
 184:             if e.in_tree && uf.find(e.f1) == big_root { tedges.push((idx, e.f1, e.f2)); }
 185:         }
 186:         let side_size = |skip: usize, start: usize| -> usize {
 187:             let mut seen = vec![false; nf];
 188:             seen[start] = true;
 189:             let mut q: VecDeque<usize> = VecDeque::new();
 190:             q.push_back(start);
 191:             let mut cnt = 1usize;
 192:             while let Some(cur) = q.pop_front() {
 193:                 for &(idx, a, b) in tedges.iter() {
 194:                     if idx == skip { continue; }
 195:                     let nb = if a == cur { b } else if b == cur { a } else { continue; };
 196:                     if !seen[nb] { seen[nb] = true; cnt += 1; q.push_back(nb); }
 197:                 }
 198:             }
 199:             cnt
 200:         };
 201:         let mut best: Option<(f64, usize)> = None;
 202:         for &(idx, f1, _f2) in tedges.iter() {
 203:             let s1 = side_size(idx, f1);
 204:             let s2 = big_len - s1;
 205:             if s1 >= MIN_PIECE_FACES && s2 >= MIN_PIECE_FACES {
 206:                 let w = dual[idx].weight;
 207:                 if best.map_or(true, |(bw, _)| w < bw) { best = Some((w, idx)); }
 208:             }
 209:         }
 210:         match best {
 211:             Some((_w, idx)) => { dual[idx].in_tree = false; }
 212:             None => { return; }
 213:         }
 214:     }
 215: }
 216:
 217: struct ForestEdge {
 218:     weight: f64,
 219:     f1: usize,
 220:     f2: usize,
 221:     va: usize,
 222:     vb: usize,
 223:     in_tree: bool,
 224: }
 225:
 226: struct Uf { parent: Vec<usize> }
 227:
 228: impl Uf {
 229:     fn new(n: usize) -> Self { Uf { parent: (0..n).collect() } }
 230:     fn find(&mut self, mut i: usize) -> usize {
 231:         let mut root = i;
 232:         while self.parent[root] != root { root = self.parent[root]; }
 233:         while self.parent[i] != root { let next = self.parent[i]; self.parent[i] = root; i = next; }
 234:         root
 235:     }
 236:     fn union(&mut self, i: usize, j: usize) -> bool {
 237:         let ri = self.find(i);
 238:         let rj = self.find(j);
 239:         if ri == rj { return false; }
 240:         self.parent[ri] = rj;
 241:         true
 242:     }
 243: }
 244:
 245: /// Точка входа papercraft-развёртки (контракт src-tauri\src\commands.rs ~L949).
 246: pub fn unfold_with_papercraft(mesh: &Mesh) -> Result<PapercraftUnfoldResult, String> {
 247:     if mesh.vertices.is_empty() || mesh.faces.is_empty() {
 248:         return Err("[MEM48-D] пустой меш: нет вершин или граней".to_string());
 249:     }
 250:     for (i, f) in mesh.faces.iter().enumerate() {
 251:         if f.vertices.iter().any(|&v| v >= mesh.vertices.len()) {
 252:             return Err(format!("[MEM48-D] грань {} ссылается на несуществующую вершину", i));
 253:         }
 254:     }
 255:
 256:     // 1. Dual-граф: общие рёбра треугольников + dihedral-веса + множество плоских рёбер.
 257:     let mut edge_face: HashMap<(usize, usize), usize> = HashMap::new();
 258:     let mut dual: Vec<ForestEdge> = Vec::new();
 259:     let mut flat_edge: HashSet<(usize, usize)> = HashSet::new();
 260:     for (fi, face) in mesh.faces.iter().enumerate() {
 261:         for k in 0..3 {
 262:             let a = face.vertices[k];
 263:             let b = face.vertices[(k + 1) % 3];
 264:             let key = if a < b { (a, b) } else { (b, a) };
 265:             if let Some(&fj) = edge_face.get(&key) {
 266:                 if edge_dihedral_deg(mesh, fi, fj) < FLAT_DEG { flat_edge.insert(key); }
 267:                 let w = dual_edge_weight(mesh, fi, fj, key.0, key.1);
 268:                 dual.push(ForestEdge { weight: w, f1: fj, f2: fi, va: key.0, vb: key.1, in_tree: false });
 269:             } else {
 270:                 edge_face.insert(key, fi);
 271:             }
 272:         }
 273:     }
 274:
 275:     // 2. Kruskal max-weight forest: дерево = каркас детали, рёбра дерева = сгибы.
 276:     dual.sort_by(|x, y| y.weight.partial_cmp(&x.weight).unwrap_or(std::cmp::Ordering::Equal));
 277:     let mut uf = Uf::new(mesh.faces.len());
 278:     for e in dual.iter_mut() {
 279:         if uf.union(e.f1, e.f2) { e.in_tree = true; }
 280:     }
 281:     // [SMART-CUT] крупные острова делим на печатабельные куски
 282:     split_oversized_islands(mesh, &mut dual);
 283:     uf = Uf::new(mesh.faces.len());
 284:     for e in dual.iter() {
 285:         if e.in_tree { uf.union(e.f1, e.f2); }
 286:     }
 287:
 288:     // 3. Компоненты леса = острова (детали).
 289:     let mut comp_map: HashMap<usize, usize> = HashMap::new();
 290:     let mut islands: Vec<Vec<usize>> = Vec::new();
 291:     for fi in 0..mesh.faces.len() {
 292:         let r = uf.find(fi);
 293:         let next = islands.len();
 294:         let id = *comp_map.entry(r).or_insert(next);
 295:         if id == next { islands.push(Vec::new()); }
 296:         islands[id].push(fi);
 297:     }
 298:
 299:     let mut tree_adj: Vec<Vec<(usize, usize, usize)>> = vec![Vec::new(); mesh.faces.len()];
 300:     let mut tree_edge_set: HashSet<(usize, usize)> = HashSet::new();
 301:     for e in &dual {
 302:         if e.in_tree {
 303:             tree_adj[e.f1].push((e.f2, e.va, e.vb));
 304:             tree_adj[e.f2].push((e.f1, e.va, e.vb));
 305:             let key = if e.va < e.vb { (e.va, e.vb) } else { (e.vb, e.va) };
 306:             tree_edge_set.insert(key);
 307:         }
 308:     }
 309:
 310:     // 4. Раскладка каждого острова.
 311:     let mut pieces: Vec<PapercraftPiece> = Vec::new();
 312:     let mut total_folds: u32 = 0;
 313:     let mut total_tabs: u32 = 0;
 314:     let mut faces_2d: Vec<[u32; 3]> = vec![[0, 0, 0]; mesh.faces.len()];
 315:     let mut verts_2d: Vec<[f64; 2]> = Vec::new();
 316:     for (island_idx, faces) in islands.iter().enumerate() {
 317:         let (piece, tabs, face2d_isl) = unfold_island(mesh, faces, island_idx as u32, &tree_edge_set, &flat_edge, &tree_adj);
 318:         for (ff, tri) in &face2d_isl { if *ff < faces_2d.len() { let base = verts_2d.len() as u32; verts_2d.push(tri[0]); verts_2d.push(tri[1]); verts_2d.push(tri[2]); faces_2d[*ff] = [base, base + 1, base + 2]; } }
 319:         total_folds += piece.fold_lines.len() as u32;
 320:         total_tabs += tabs as u32;
 321:         pieces.push(piece);
 322:     }
 323:
 324:     let complexity = (pieces.len() as f32) * 2.0 + (total_folds as f32) * 0.5 + (total_tabs as f32) * 0.1;
 325:     let metrics = UnwrapMetrics {
 326:         total_pieces: pieces.len() as u32,
 327:         total_pages: 0,
 328:         total_folds,
 329:         total_tabs,
 330:         complexity_score: complexity.min(100.0),
 331:     };
 332:     let mut face_pieces: Vec<u32> = vec![0; mesh.faces.len()];
 333:     for (ii, faces) in islands.iter().enumerate() { for &f in faces { if f < face_pieces.len() { face_pieces[f] = ii as u32; } } }
 334:     Ok(PapercraftUnfoldResult { pieces, metrics, parts: Vec::new(), face_pieces, vertices_2d: verts_2d, faces_2d, engine_used: "MST-Seam (papercraft)".to_string() })
 335: }
 336:
 337: fn unfold_island(
 338:     mesh: &Mesh,
 339:     island_faces: &[usize],
 340:     piece_id: u32,
 341:     tree_edge_set: &HashSet<(usize, usize)>,
 342:     flat_edge: &HashSet<(usize, usize)>,
 343:     tree_adj: &[Vec<(usize, usize, usize)>],
 344: ) -> (PapercraftPiece, usize, HashMap<usize, [[f64; 2]; 3]>) {
 345:     // 2D-позиции храним ПО ГРАНЯМ (3 угла каждой грани в своей рамке), а не по
 346:     // вершинам: у реза две стороны, их концы обязаны расходиться в развёртке.
 347:     let mut face2d: HashMap<usize, [[f64; 2]; 3]> = HashMap::new();
 348:
 349:     // 4a. Корневая грань: v0 в (0,0), ребро v0->v1 вдоль +X, v2 по шарниру.
 350:     let root = island_faces[0];
 351:     let rf = &mesh.faces[root];
 352:     let (r0, r1, r2) = (rf.vertices[0], rf.vertices[1], rf.vertices[2]);
 353:     let base = len3(sub3(mesh.vertices[r1].position, mesh.vertices[r0].position));
 354:     let r2p = unfold_hinge(mesh.vertices[r0].position, mesh.vertices[r1].position, mesh.vertices[r2].position, [0.0, 0.0], [base, 0.0], 1.0);
 355:     face2d.insert(root, [[0.0, 0.0], [base, 0.0], r2p]);
 356:
 357:     // 4b. BFS по дереву: каждая новая грань ложится шарниром на уже разложенное ребро.
 358:     let mut visited: HashSet<usize> = HashSet::new();
 359:     visited.insert(root);
 360:     let mut queue: VecDeque<usize> = VecDeque::new();
 361:     queue.push_back(root);
 362:     while let Some(cur) = queue.pop_front() {
 363:         let cpos = *face2d.get(&cur).unwrap_or(&[[0.0, 0.0]; 3]);
 364:         let cf = &mesh.faces[cur];
 365:         for &(nb, va, vb) in &tree_adj[cur] {
 366:             if visited.contains(&nb) { continue; }
 367:             visited.insert(nb);
 368:             let ia = cf.vertices.iter().position(|&v| v == va).unwrap_or(0);
 369:             let ib = cf.vertices.iter().position(|&v| v == vb).unwrap_or(1);
 370:             let a2 = cpos[ia];
 371:             let b2 = cpos[ib];
 372:             let nf = &mesh.faces[nb];
 373:             let vc = *nf.vertices.iter().find(|&&v| v != va && v != vb).unwrap_or(&va);
 374:             let a3 = mesh.vertices[va].position;
 375:             let b3 = mesh.vertices[vb].position;
 376:             let c3 = mesh.vertices[vc].position;
 377:             let ja = nf.vertices.iter().position(|&v| v == va).unwrap_or(0);
 378:             let jb = nf.vertices.iter().position(|&v| v == vb).unwrap_or(1);
 379:             let jc = nf.vertices.iter().position(|&v| v == vc).unwrap_or(2);
 380:             // [MEM48-E] Знак разворота: третья вершина кладётся слева от направления
 381:             // шарнира В ОБХОДЕ НОВОЙ грани (все грани выходят CCW, как корень).
 382:             // Старый код всегда клал слева от va->vb, поэтому каждая вторая грань
 383:             // зеркалилась через шарнир и развёртка самоперекрывалась
 384:             // (на кубе outline area 4 != 6.0).
 385:             let sign = if jb == (ja + 1) % 3 { 1.0 } else { -1.0 };
 386:             let c2 = unfold_hinge(a3, b3, c3, a2, b2, sign);
 387:             let mut npos = [[0.0, 0.0]; 3];
 388:             npos[ja] = a2;
 389:             npos[jb] = b2;
 390:             npos[jc] = c2;
 391:             face2d.insert(nb, npos);
 392:             queue.push_back(nb);
 393:         }
 394:     }
 395:
 396:     let face_pos = |fi: usize, v: usize| -> [f64; 2] {
 397:         let p = *face2d.get(&fi).unwrap_or(&[[0.0, 0.0]; 3]);
 398:         let f = &mesh.faces[fi];
 399:         let i = f.vertices.iter().position(|&x| x == v).unwrap_or(0);
 400:         p[i]
 401:     };
 402:
 403:     // 4c. Сгибы = рёбра дерева внутри острова, КРОМЕ плоских (SoftHidden в craft.rs).
 404:     let mut fold_lines: Vec<[[f64; 2]; 2]> = Vec::new();
 405:     let mut fold_seen: HashSet<(usize, usize)> = HashSet::new();
 406:     for &fi in island_faces {
 407:         let f = &mesh.faces[fi];
 408:         for k in 0..3 {
 409:             let a = f.vertices[k];
 410:             let b = f.vertices[(k + 1) % 3];
 411:             let key = if a < b { (a, b) } else { (b, a) };
 412:             if tree_edge_set.contains(&key) && !flat_edge.contains(&key) && fold_seen.insert(key) {
 413:                 fold_lines.push([face_pos(fi, a), face_pos(fi, b)]);
 414:             }
 415:         }
 416:     }
 417:
 418:     // 4d. Граница = направленные рёбра граней острова, не входящие в дерево
 419:     // (внешний контур + обе стороны внутренних резов), с указанием грани-владельца.
 420:     let mut boundary: Vec<(usize, usize, usize)> = Vec::new(); // (грань, a, b)
 421:     for &fi in island_faces {
 422:         let f = &mesh.faces[fi];
 423:         for k in 0..3 {
 424:             let a = f.vertices[k];
 425:             let b = f.vertices[(k + 1) % 3];
 426:             let key = if a < b { (a, b) } else { (b, a) };
 427:             if !tree_edge_set.contains(&key) { boundary.push((fi, a, b)); }
 428:         }
 429:     }
 430:
 431:     // 4e. Внешний контур: обход направленных рёбер границы по 2D-ПОЗИЦИЯМ,
 432:     // а не по id вершин. Все грани уложены CCW (знак в unfold_hinge), поэтому
 433:     // у каждого граничного ребра внутренность слева; на "щипковых" вершинах,
 434:     // где контур касается себя, продолжение выбирается самым крутым поворотом
 435:     // ПО часовой стрелке - это трассировка внешней грани: одна петля покрывает
 436:     // ВСЮ границу (восьмёрка на щипке), площадь по шнурку = площадь детали.
 437:     let mut outline_2d: Vec<[f64; 2]> = Vec::new();
 438:     {
 439:         let mut segs: Vec<([f64; 2], [f64; 2])> = Vec::new();
 440:         for &(fi, a, b) in &boundary {
 441:             let pa = face_pos(fi, a);
 442:             let pb = face_pos(fi, b);
 443:             if len2(sub2(pb, pa)) > 1e-12 { segs.push((pa, pb)); }
 444:         }
 445:         let key = |p: [f64; 2]| -> (i64, i64) {
 446:             ((p[0] / 1e-6).round() as i64, (p[1] / 1e-6).round() as i64)
 447:         };
 448:         let mut by_start: HashMap<(i64, i64), Vec<usize>> = HashMap::new();
 449:         for (i, s) in segs.iter().enumerate() {
 450:             by_start.entry(key(s.0)).or_default().push(i);
 451:         }
 452:         let mut used = vec![false; segs.len()];
 453:         let mut best_area = 0.0f64;
 454:         for start in 0..segs.len() {
 455:             if used[start] { continue; }
 456:             let mut poly: Vec<[f64; 2]> = Vec::new();
 457:             let mut cur = start;
 458:             let mut guard = 0usize;
 459:             while guard <= segs.len() {
 460:                 guard += 1;
 461:                 used[cur] = true;
 462:                 poly.push(segs[cur].0);
 463:                 let din = sub2(segs[cur].1, segs[cur].0);
 464:                 let mut nxt: Option<usize> = None;
 465:                 let mut best_t = 0.0f64;
 466:                 if let Some(cands) = by_start.get(&key(segs[cur].1)) {
 467:                     for &j in cands {
 468:                         if used[j] { continue; }
 469:                         let dj = sub2(segs[j].1, segs[j].0);
 470:                         let t = (din[0] * dj[1] - din[1] * dj[0]).atan2(dot2(din, dj));
 471:                         if nxt.is_none() || t < best_t { best_t = t; nxt = Some(j); }
 472:                     }
 473:                 }
 474:                 match nxt {
 475:                     Some(j) => { cur = j; }
 476:                     None => { break; }
 477:                 }
 478:             }
 479:             let mut sa = 0.0f64;
 480:             let n = poly.len();
 481:             for i in 0..n {
 482:                 let q = (i + 1) % n;
 483:                 sa += poly[i][0] * poly[q][1] - poly[q][0] * poly[i][1];
 484:             }
 485:             if sa.abs() > best_area { best_area = sa.abs(); outline_2d = poly; }
 486:         }
 487:     }
 488:
 489:     // 4f. Клапаны (FlapSide/flap_double=false: ОДИН клапан на рез, трапеция наружу
 490:     // от центроида; вторая сторона реза остаётся гладкой).
 491:     let mut cut_lines: Vec<[[f64; 2]; 2]> = Vec::new();
 492:     let mut tabbed: HashSet<(usize, usize)> = HashSet::new();
 493:     let mut tabs = 0usize;
 494:     let mut cx = 0.0;
 495:     let mut cy = 0.0;
 496:     let mut cn = 0usize;
 497:     for &fi in island_faces {
 498:         if let Some(p) = face2d.get(&fi) {
 499:             for c in p.iter() { cx += c[0]; cy += c[1]; cn += 1; }
 500:         }
 501:     }
 502:     let centroid = if cn == 0 { [0.0, 0.0] } else { [cx / cn as f64, cy / cn as f64] };
 503:
 504:     let mut avg_len = 0.0;
 505:     for &(fi, a, b) in &boundary {
 506:         let pa = face_pos(fi, a);
 507:         let pb = face_pos(fi, b);
 508:         avg_len += len2(sub2(pb, pa));
 509:     }
 510:     let avg_len = if boundary.is_empty() { 0.0 } else { avg_len / boundary.len() as f64 };
 511:     let tab_default = 0.3 * avg_len;
 512:
 513:     // [FIX60B] митры по углам границы: у вершины стыка двух граничных рёбер
 514:     // внешний угол ω на стороне клапана; при ω < 180° верх трапеции отодвигается
 515:     // от вершины на t >= tw/tan(ω/2) — иначе соседние клапаны накладываются.
 516:     let mut vtx_edges: HashMap<usize, Vec<usize>> = HashMap::new();
 517:     for (ei, &(_, va, vb)) in boundary.iter().enumerate() {
 518:         vtx_edges.entry(va).or_insert_with(Vec::new).push(ei);
 519:         vtx_edges.entry(vb).or_insert_with(Vec::new).push(ei);
 520:     }
 521:     for (ei, &(fi, a, b)) in boundary.iter().enumerate() {
 522:         let pa = face_pos(fi, a);
 523:         let pb = face_pos(fi, b);
 524:         let elen = len2(sub2(pb, pa));
 525:         if elen < 1e-9 { cut_lines.push([pa, pb]); continue; }
 526:         let key = if a < b { (a, b) } else { (b, a) };
 527:         if tabbed.contains(&key) { cut_lines.push([pa, pb]); continue; }
 528:         tabbed.insert(key);
 529:         let d = scale2(sub2(pb, pa), 1.0 / elen);
 530:         let mut n = [d[1], -d[0]];
 531:         let mid = scale2(add2(pa, pb), 0.5);
 532:         if dot2(sub2(mid, centroid), n) < 0.0 { n = [-n[0], -n[1]]; }
 533:         let mut tw = tab_default.min(0.45 * elen);
 534:         let inset = 0.3 * tw;
 535:         let mut t_fit = [inset, inset];
 536:         let mut oms = [std::f64::consts::PI, std::f64::consts::PI];
 537:         let ends = [(a, pa, 0usize), (b, pb, 1usize)];
 538:         for &(v, p_v, k) in ends.iter() {
 539:             let evs = match vtx_edges.get(&v) { Some(x) => x, None => continue };
 540:             if evs.len() != 2 { continue; }
 541:             let oi = if evs[0] == ei { evs[1] } else { evs[0] };
 542:             let (fj, c1, c2) = boundary[oi];
 543:             let v2 = if c1 == v { c2 } else { c1 };
 544:             let p2 = face_pos(fj, v2);
 545:             let dv = sub2(p2, p_v);
 546:             let lv = len2(dv);
 547:             if lv < 1e-9 { continue; }
 548:             let d2 = scale2(dv, 1.0 / lv);
 549:             let d1 = if v == a { d } else { scale2(d, -1.0) };
 550:             let cr = d1[0] * d2[1] - d1[1] * d2[0];
 551:             let cs = dot2(d1, d2);
 552:             let sgn = d[0] * n[1] - d[1] * n[0];
 553:             let ang = cr.atan2(cs);
 554:             let tau = std::f64::consts::TAU;
 555:             let om = if sgn < 0.0 { if ang < 0.0 { -ang } else { tau - ang } } else { if ang > 0.0 { ang } else { tau + ang } };
 556:             oms[k] = om;
 557:             if om < std::f64::consts::PI && om > 1e-6 {
 558:                 let need = tw / (om * 0.5).tan();
 559:                 if need > t_fit[k] { t_fit[k] = need.min(0.45 * elen); }
 560:             }
 561:         }
 562:         for k in 0..2usize {
 563:             if oms[k] < std::f64::consts::PI && t_fit[k] >= 0.45 * elen - 1e-9 {
 564:                 let tw2 = t_fit[k] * (oms[k] * 0.5).tan() * 0.95;
 565:                 if tw2 < tw { tw = tw2; }
 566:             }
 567:         }
 568:         let top1 = add2(add2(pa, scale2(n, tw)), scale2(d, t_fit[0]));
 569:         let top2 = add2(add2(pb, scale2(n, tw)), scale2(d, -t_fit[1]));
 570:         cut_lines.push([pa, top1]);
 571:         cut_lines.push([top1, top2]);
 572:         cut_lines.push([top2, pb]);
 573:         tabs += 1;
 574:     }

 576:     // 4g. Контур построен в 4e (обход границы по 2D-позициям с выбором
 577:     // поворота) - здесь остаётся только сборка piece.
 578:
 579:     let face_ids: Vec<u32> = island_faces.iter().map(|&f| f as u32).collect();
 580:     let piece = PapercraftPiece { id: piece_id, face_ids, outline_2d, fold_lines, cut_lines, part_id: 0, part: String::new() }; // [82-A] part заполнит unfold_with_papercraft_parts
 581:     (piece, tabs, face2d)

 589: /// Направление ребра (va,vb) в обходе грани fi: vb-va, если грань идёт va->vb,
 590: /// иначе va-vb. Нужен для ЗНАКА вогнутости.
 591: fn edge_dir_in_face(mesh: &Mesh, fi: usize, va: usize, vb: usize) -> [f64; 3] {
 592:     let f = &mesh.faces[fi];
 593:     for k in 0..3 {
 594:         if f.vertices[k] == va && f.vertices[(k + 1) % 3] == vb {
 595:             return sub3(mesh.vertices[vb].position, mesh.vertices[va].position);
 596:         }
 597:     }
 598:     sub3(mesh.vertices[va].position, mesh.vertices[vb].position)
 599: }

 601: /// Знаковая вогнутость ребра: >0 выпуклое, <0 вогнутое (модуль ~ sin угла складки).
 602: /// Знак инвариантен к тому, какую из двух граней считать первой.
 603: fn signed_concavity(mesh: &Mesh, f1: usize, f2: usize, va: usize, vb: usize) -> f64 {
 604:     let n1 = face_normal(mesh, f1);
 605:     let n2 = face_normal(mesh, f2);
 606:     let d = edge_dir_in_face(mesh, f1, va, vb);
 607:     let l = len3(d);
 608:     if l < 1e-12 { return 0.0; }
 609:     let cx = n1[1] * n2[2] - n1[2] * n2[1];
 610:     let cy = n1[2] * n2[0] - n1[0] * n2[2];
 611:     let cz = n1[0] * n2[1] - n1[1] * n2[0];
 612:     (cx * d[0] + cy * d[1] + cz * d[2]) / l
 613: }

 615: /// Авто-сегментация: грани делятся на части разрезами по вогнутым складкам
 616: /// (складка внутрь глубже concave_deg) и по смене материала. Обрывки <3 граней
 617: /// присоединяются к соседней части.
 618: fn segment_auto(mesh: &Mesh, concave_deg: f64) -> Vec<Vec<usize>> {
 619:     let nf = mesh.faces.len();
 620:     let sin_th = concave_deg.to_radians().sin();
 621:     let mut edge_face: HashMap<(usize, usize), usize> = HashMap::new();
 622:     let mut adj: Vec<Vec<usize>> = vec![Vec::new(); nf];
 623:     for fi in 0..nf {
 624:         let face = &mesh.faces[fi];
 625:         for k in 0..3 {
 626:             let a = face.vertices[k];
 627:             let b = face.vertices[(k + 1) % 3];
 628:             let key = if a < b { (a, b) } else { (b, a) };
 629:             if let Some(&fj) = edge_face.get(&key) {
 630:                 let mut do_cut = signed_concavity(mesh, fj, fi, key.0, key.1) < -sin_th;
 631:                 if let (Some(x), Some(y)) = (mesh.faces[fj].material_id, mesh.faces[fi].material_id) {
 632:                     if x != y { do_cut = true; }
 633:                 }
 634:                 if !do_cut {
 635:                     adj[fj].push(fi);
 636:                     adj[fi].push(fj);
 637:                 }
 638:             } else {
 639:                 edge_face.insert(key, fi);
 640:             }
 641:         }
 642:     }
 643:     // компоненты связности без разрезов = черновые части
 644:     let mut comp_of = vec![usize::MAX; nf];
 645:     let mut comps: Vec<Vec<usize>> = Vec::new();
 646:     for start in 0..nf {
 647:         if comp_of[start] != usize::MAX { continue; }
 648:         let id = comps.len();
 649:         let mut stack = vec![start];
 650:         comp_of[start] = id;
 651:         let mut members: Vec<usize> = Vec::new();
 652:         while let Some(cur) = stack.pop() {
 653:             members.push(cur);
 654:             for &nb in &adj[cur] {
 655:                 if comp_of[nb] == usize::MAX {
 656:                     comp_of[nb] = id;
 657:                     stack.push(nb);
 658:                 }
 659:             }
 660:         }
 661:         comps.push(members);
 662:     }
 663:     // обрывки (<3 граней) вливаем в наибольшего соседа
 664:     let mut alive: Vec<bool> = vec![true; comps.len()];
 665:     for i in 0..comps.len() {
 666:         if comps[i].len() >= 3 || !alive[i] { continue; }
 667:         let mut best: Option<usize> = None;
 668:         let mut best_n = 0usize;
 669:         for &f in &comps[i] {
 670:             for &nb in &adj[f] {
 671:                 let cj = comp_of[nb];
 672:                 if cj == i || cj >= comps.len() || !alive[cj] { continue; }
 673:                 if comps[cj].len() > best_n { best_n = comps[cj].len(); best = Some(cj); }
 674:             }
 675:         }
 676:         if let Some(j) = best {
 677:             let frag: Vec<usize> = comps[i].clone();
 678:             comps[j].extend(frag);
 679:             alive[i] = false;
 680:         }
 681:     }
 682:     let mut parts: Vec<Vec<usize>> = Vec::new();
 683:     for i in 0..comps.len() {
 684:         if alive[i] && !comps[i].is_empty() { parts.push(comps[i].clone()); }
 685:     }
 686:     parts.sort_by(|x, y| y.len().partial_cmp(&x.len()).unwrap_or(std::cmp::Ordering::Equal));
 687:     parts
 688: }

 690: /// [82-A] Развёртка по частям. Каждая часть разворачивается независимо,
 691: /// piece получает part_id/part, face_ids переводятся в индексы исходного меша.
 692: pub fn unfold_with_papercraft_parts(mesh: &Mesh, cfg: &PartsConfig) -> Result<PapercraftUnfoldResult, String> {
 693:     let nf = mesh.faces.len();
 694:     if mesh.vertices.is_empty() || nf == 0 {
 695:         return Err("[82-A] пустой меш: нет вершин или граней".to_string());
 696:     }
 697:     let mut face_lists: Vec<Vec<usize>> = Vec::new();
 698:     let mut names: Vec<String> = Vec::new();
 699:     let mut source = "whole".to_string();
 700:     match cfg.mode.as_str() {
 701:         "materials" => {
 702:             source = "material".to_string();
 703:             let mut by_mat: HashMap<usize, Vec<usize>> = HashMap::new();
 704:             let mut none_faces: Vec<usize> = Vec::new();
 705:             for (fi, f) in mesh.faces.iter().enumerate() {
 706:                 match f.material_id {
 707:                     Some(m) => { by_mat.entry(m).or_default().push(fi); }
 708:                     None => { none_faces.push(fi); }
 709:                 }
 710:             }
 711:             if by_mat.is_empty() {
 712:                 face_lists.push((0..nf).collect());
 713:                 names.push("Модель".to_string());
 714:                 source = "whole".to_string();
 715:             } else {
 716:                 let mut keys: Vec<usize> = by_mat.keys().cloned().collect();
 717:                 keys.sort();
 718:                 for m in keys {
 719:                     face_lists.push(by_mat[&m].clone());
 720:                     let nm = mesh.metadata.material_names.get(m).cloned().unwrap_or_else(|| format!("Материал {}", m));
 721:                     names.push(nm);
 722:                 }
 723:                 if !none_faces.is_empty() {
 724:                     face_lists.push(none_faces);
 725:                     names.push("Без материала".to_string());
 726:                 }
 727:             }
 728:         }
 729:         "auto" => {
 730:             source = "auto".to_string();
 731:             let seg = segment_auto(mesh, cfg.concave_deg);
 732:             if seg.len() > 1 {
 733:                 for (i, s) in seg.iter().enumerate() {
 734:                     face_lists.push(s.clone());
 735:                     names.push(format!("Часть {}", i + 1));
 736:                 }
 737:             } else {
 738:                 // складок не нашли - вся модель одной частью (как раньше)
 739:                 face_lists.push((0..nf).collect());
 740:                 names.push("Модель".to_string());
 741:                 source = "whole".to_string();
 742:             }
 743:         }
 744:         "manual" => {
 745:             // [82-A] задел под вариант 3: пользователь сам размечает части
 746:             source = "manual".to_string();
 747:             for (i, list) in cfg.manual.iter().enumerate() {
 748:                 let faces: Vec<usize> = list.iter().filter(|&&f| (f as usize) < nf).map(|&f| f as usize).collect();
 749:                 if faces.is_empty() { continue; }
 750:                 face_lists.push(faces);
 751:                 names.push(format!("Ручная {}", i + 1));
 752:             }
 753:             if face_lists.is_empty() {
 754:                 return Err("[82-A] manual: нет валидных списков граней".to_string());
 755:             }
 756:         }
 757:         _ => {
 758:             face_lists.push((0..nf).collect());
 759:             names.push("Модель".to_string());
 760:         }
 761:     }
 762:
 763:     let mut pieces: Vec<PapercraftPiece> = Vec::new();
 764:     let mut parts_meta: Vec<ModelPart> = Vec::new();
 765:     let mut total_folds = 0u32;
 766:     let mut total_tabs = 0u32;
 767:     let mut faces_2d: Vec<[u32; 3]> = vec![[0, 0, 0]; mesh.faces.len()];
 768:     let mut verts_2d: Vec<[f64; 2]> = Vec::new();
 769:     for (pi, faces) in face_lists.iter().enumerate() {
 770:         if faces.is_empty() { continue; }
 771:         let mut sub = Mesh::new(&names[pi]);
 772:         for v in &mesh.vertices { sub.add_vertex(v.clone()); }
 773:         let mut orig_of_sub: Vec<usize> = Vec::with_capacity(faces.len());
 774:         for &fi in faces {
 775:             sub.add_face(mesh.faces[fi].clone());
 776:             orig_of_sub.push(fi);
 777:         }
 778:         let res = unfold_with_papercraft(&sub)?;
 779:         total_folds += res.metrics.total_folds;
 780:         total_tabs += res.metrics.total_tabs;
 781:         let off = verts_2d.len() as u32;
 782:         for p in &res.vertices_2d { verts_2d.push(*p); }
 783:         for (ff, tri) in res.faces_2d.iter().enumerate() {
 784:             if ff < orig_of_sub.len() && !(tri[0] == tri[1] && tri[1] == tri[2]) {
 785:                 let of = orig_of_sub[ff];
 786:                 if of < faces_2d.len() { faces_2d[of] = [off + tri[0], off + tri[1], off + tri[2]]; }
 787:             }
 788:         }
 789:         for mut piece in res.pieces {
 790:             piece.face_ids = piece.face_ids.iter().map(|&f| orig_of_sub[f as usize] as u32).collect();
 791:             piece.part_id = pi as u32;
 792:             piece.part = names[pi].clone();
 793:             piece.id = pieces.len() as u32;
 794:             pieces.push(piece);
 795:         }
 796:         parts_meta.push(ModelPart { id: pi as u32, name: names[pi].clone(), source: source.clone(), face_count: faces.len() });
 797:     }
 798:
 799:     let complexity = (pieces.len() as f32) * 2.0 + (total_folds as f32) * 0.5 + (total_tabs as f32) * 0.1;
 800:     let metrics = UnwrapMetrics {
 801:         total_pieces: pieces.len() as u32,
 802:         total_pages: 0,
 803:         total_folds,
 804:         total_tabs,
 805:         complexity_score: complexity.min(100.0),
 806:     };
 807:     let mut face_pieces: Vec<u32> = vec![0; nf];
 808:     for p in &pieces { for &f in &p.face_ids { if (f as usize) < nf { face_pieces[f as usize] = p.id; } } }
 809:     Ok(PapercraftUnfoldResult { pieces, metrics, parts: parts_meta, face_pieces, vertices_2d: verts_2d, faces_2d, engine_used: format!("MST-Seam + {} parts", cfg.mode) })
 810: }
