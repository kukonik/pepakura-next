```typescript
// Пример интеграции новой системы загрузки с Three.js сценой
import { NewUniversalModelLoader } from './loaders/NewUniversalModelLoader';
import { TransformationController } from './modules/renderer-3d/TransformationController';
import { MaterialProcessor } from './modules/materials/MaterialProcessor';

class SceneIntegration {
  private modelLoader: NewUniversalModelLoader;
  private transformationController: TransformationController;
  private materialProcessor: MaterialProcessor;
  private scene: THREE.Scene;
  
  constructor(scene: THREE.Scene) {
    this.scene = scene;
    this.modelLoader = new NewUniversalModelLoader();
    this.transformationController = new TransformationController();
    this.materialProcessor = new MaterialProcessor();
  }
  
  async loadModel(file: File): Promise<void> {
    try {
      // Загрузка модели
      const loadResult = await this.modelLoader.load(file);
      
      // Обработка материалов
      if (loadResult.materials) {
        const processedMaterials = await this.materialProcessor.process(loadResult.materials);
        // Применение материалов к объекту
      }
      
      // Добавление объекта в сцену
      this.scene.add(loadResult.object);
      
      // Установка объекта для трансформаций
      this.transformationController.setObject(loadResult.object);
    } catch (error) {
      console.error('Ошибка загрузки модели:', error);
      throw error;
    }
  }
}
```

### Система Undo/Redo

```typescript
// Пример архитектуры системы Undo/Redo
interface Command {
  execute(): void;
  undo(): void;
  getDescription(): string;
}

class CommandManager {
  private history: Command[] = [];
  private currentIndex: number = -1;
  private maxSize: number = 50;
  
  executeCommand(command: Command): void {
    // Удаление отмененных команд
    this.history = this.history.slice(0, this.currentIndex + 1);
    
    // Выполнение команды
    command.execute();
    this.history.push(command);
    
    // Ограничение размера истории
    if (this.history.length > this.maxSize) {
      this.history.shift();
    } else {
      this.currentIndex++;
    }
  }
  
  undo(): boolean {
    if (this.currentIndex >= 0) {
      const command = this.history[this.currentIndex];
      command.undo();
      this.currentIndex--;
      return true;
    }
    return false;
  }
  
  redo(): boolean {
    if (this.currentIndex < this.history.length - 1) {
      this.currentIndex++;
      const command = this.history[this.currentIndex];
      command.execute();
      return true;
    }
    return false;
  }
}
```
